  try{

function cloneBlock(id){
  if(!id)return toast('请先选中要复制的内容块');
  const orig=blocks.find(b=>b.id===id);
  if(!orig)return;
  const copy=JSON.parse(JSON.stringify(orig));
  copy.id=genId();
  if(copy.type==='chart')copy.id=copy.id;
  const origIdx=blocks.findIndex(b=>b.id===id);
  blocks.splice(origIdx+1,0,copy);
  selectBlock(copy.id);
  renderAll();
  autoSave();
  toast('已复制');
}

  if(selectedBlockId===id)selectedBlockId=null;

  autoSave();
  toast('已删除');
}

  const sec=sections.find(s=>s.id===id);
  if(!sec)return;
  if(sections.length<=1){toast('至少保留 1 个分区');return;}
  if(!confirm('确认删除分区「'+sec.name+'」？该分区下的内容块也会一并删除。'))return;

// 渲染表格块（支持多层表头 + 单元格合并 + 手动标记）
function renderTableBlock(block){
  const title=escHtml(block.title||'');
  const headers=block.headers||[];
  const rows=block.rows||[];
  // 表头行数（1-3）
  const headerRows=block.headerRows||1;
  // 多层表头数据（如果定义了 headerLevels，则按其渲染；否则回退到单一表头）
  const headerLevels=block.headerLevels||null;
  // 合并单元格映射：{ "r,c": {rowspan,colspan} }
  const merges=block.merges||{};
  // 手动标记：cellMarks["ri,ci"] / rowMarks["ri"] / colMarks["ci"] → 'leader'|'lowest'|'warning'|'critical'|'info'|'success'|'muted'
  const cellMarks=block.cellMarks||{};
  const rowMarks=block.rowMarks||{};
  const colMarks=block.colMarks||{};
  // 渲染表头
  let headHtml='';
  if(headerLevels&&headerLevels.length>0){
    // 多层表头模式
    for(let hr=0;hr<headerLevels.length;hr++){
      headHtml+='<tr>';
      headerLevels[hr].forEach((cell,ci)=>{
        const key=hr+','+ci;
        const m=merges[key];
        let attrs='';
        let txt=cell;
        if(typeof cell==='object'&&cell!==null){
          txt=cell.text||'';
          if(cell.rowspan)attrs+=' rowspan="'+cell.rowspan+'"';
          if(cell.colspan)attrs+=' colspan="'+cell.colspan+'"';
        }else{
          if(m){
            if(m.rowspan>1)attrs+=' rowspan="'+m.rowspan+'"';
            if(m.colspan>1)attrs+=' colspan="'+m.colspan+'"';
          }
        }
        headHtml+='<th class="merged-cell"'+attrs+'>'+escHtml(txt)+'</th>';
      });
      headHtml+='</tr>';
    }
  }else{
    // 单层表头（兼容老数据）
    headHtml+='<tr>'+headers.map((h,i)=>{
      const m=merges['0,'+i];
      let attrs='';
      if(m){
        if(m.rowspan>1)attrs+=' rowspan="'+m.rowspan+'"';
        if(m.colspan>1)attrs+=' colspan="'+m.colspan+'"';
      }
      return '<th class="merged-cell"'+attrs+'>'+escHtml(h)+'</th>';
    }).join('')+'</tr>';
  }
  // 渲染表格体（含手动标记）
  let bodyHtml='';
  rows.forEach((row,ri)=>{
    bodyHtml+='<tr>';
    for(let ci=0;ci<headers.length;ci++){
      const mkey=ri+headerRows+','+ci; // 表头行偏移
      const m=merges[mkey];
      let attrs='';
      if(m){
        if(m.rowspan>1)attrs+=' rowspan="'+m.rowspan+'"';
        if(m.colspan>1)attrs+=' colspan="'+m.colspan+'"';
        // 被合并的格子需要 skip
        if(m.skip)continue;
      }
      // 手动标记优先级：cell > row > col
      const mark=cellMarks[ri+','+ci]||rowMarks[ri]||colMarks[ci]||'';
      let cellContent=escHtml(row[ci]!==undefined?row[ci]:'');
      if(mark){
        cellContent='<span class="cell-mark-'+mark+'">'+cellContent+'</span>';
      }
      // 若当前单元格被某条 rowMarks / colMarks / cellMarks 标记，给 td 加同 class（覆盖整格背景）
      let tdClass='';
      if(cellMarks[ri+','+ci])tdClass=' cell-mark cell-mark-'+cellMarks[ri+','+ci];
      else if(rowMarks[ri])tdClass=' cell-mark cell-mark-row-'+rowMarks[ri];
      else if(colMarks[ci])tdClass=' cell-mark cell-mark-col-'+colMarks[ci];
      bodyHtml+='<td'+attrs+tdClass+'>'+cellContent+'</td>';
    }
    bodyHtml+='</tr>';
  });
  return '<h3 style="margin-bottom:8px">'+title+'</h3><div class="table-wrap"><table><thead>'+headHtml+'</thead><tbody>'+bodyHtml+'</tbody></table></div>';
}

function renderChartBlock(block){
  const canvas=document.getElementById(block.id);
  if(!canvas)return;
  if(chartInstances[block.id])chartInstances[block.id].destroy();
  const type=block.type_chart||(block.type!=='chart'?block.type:'bar')||'bar';
  const labels=block.labels||[];
  // 判断是否为组合图（任一系列指定了 seriesType=line 而 type=bar，或任意 dataset 有自己的 seriesType 与全局不同）
  const isCombo=type==='combo'||(type==='bar'&&(block.datasets||[]).some(ds=>ds.seriesType==='line'));
  const effectiveType=isCombo?'bar':type;
  const datasets=(block.datasets||[]).map((ds,i)=>{
    const bg=ds.backgroundColor||getColor(i);
    const seriesType=(isCombo?(ds.seriesType||'bar'):type);
    return {
      type:seriesType,
      label:ds.label||'系列'+(i+1),
      data:ds.data||[],
      backgroundColor:(seriesType==='pie'||seriesType==='doughnut'||seriesType==='polarArea')?getColorPalette((ds.data||[]).length):bg,
      borderColor:seriesType==='line'?bg:undefined,
      borderWidth:seriesType==='line'?(ds.borderWidth||2.5):(seriesType==='bar'?(ds.borderWidth||0):undefined),
      borderRadius:seriesType==='bar'?(ds.borderRadius!==undefined?ds.borderRadius:6):undefined,
      tension:seriesType==='line'?(ds.tension!==undefined?ds.tension:0.3):undefined,
      fill:seriesType==='line'?(ds.fill!==undefined?ds.fill:false):undefined,
      pointRadius:seriesType==='line'?(ds.pointRadius!==undefined?ds.pointRadius:4):undefined,
      pointHoverRadius:seriesType==='line'?6:undefined,
      yAxisID:ds.yAxisID||'y',
      borderDash:ds.borderDash||undefined,
      datalabels:seriesType==='line'?{display:false}:undefined
    };
  });
  // 是否存在第二 Y 轴系列
  const hasY2=(block.datasets||[]).some(ds=>(ds.yAxisID||'y')==='y2');
  const cfg={type:effectiveType,data:{labels,datasets},options:{responsive:true,maintainAspectRatio:false,plugins:{legend:{position:'bottom',labels:{padding:16,usePointStyle:true,font:{size:13}}}}}};
  if(effectiveType==='bar'||effectiveType==='line'||effectiveType==='radar'){
    cfg.options.scales={
      x:{ticks:{font:{size:11}},grid:{display:false}},
      y:{ticks:{font:{size:11}},position:'left',title:{display:hasY2,text:'主Y轴',font:{size:11}}}
    };
    if(hasY2){
      cfg.options.scales.y2={
        type:'linear',position:'right',
        ticks:{font:{size:11},color:'#e65100'},
        title:{display:true,text:'副Y轴',font:{size:11,color:'#e65100'}},
        grid:{drawOnChartArea:false}
      };
    }
  }
  if(type==='pie'||type==='doughnut'||type==='polarArea'){
    cfg.options.plugins.datalabels={display:true,color:'#fff',font:{size:12,weight:'bold'},formatter:function(v,c){var t=c.chart.data.datasets[0].data.reduce(function(a,b){return a+b;},0);return t>0?(v/t*100).toFixed(1)+'%':'';}};
  }
  chartInstances[block.id]=new Chart(canvas,cfg);
}

function liveResizeBlock(id,prop,value){
  const el=document.querySelector('.block[data-block-id="'+id+'"]');
  if(!el)return;
  const v=parseInt(value)||0;
  const block=blocks.find(b=>b.id===id);
  if(prop==='blockWidth'){
    el.style.width=v===100?'100%':'calc('+v+'% - 6px)';
  }else{
    if(v===0){el.style.maxHeight='';el.style.overflowY='';}
    else{el.style.maxHeight=v+'px';el.style.overflowY='auto';}
    if(block&&block.type==='chart'){
      const wrap=el.querySelector('.chart-canvas-wrap');
      if(wrap)wrap.style.height=v>0?v+'px':'400px';
      if(chartInstances[id])setTimeout(function(){if(chartInstances[id])chartInstances[id].resize();},10);
    }
  }
  // 锁定比例联动
  const otherProp=syncLockedDimension(id,prop,value);
  if(otherProp&&block){
    if(otherProp==='blockWidth'){
      el.style.width=(block.blockWidth||100)===100?'100%':'calc('+(block.blockWidth||100)+'% - 6px)';
    }else{
      const oh=block.blockHeight;
      if(oh&&oh!==0&&oh!=='0'&&oh!=='auto'){el.style.maxHeight=oh+'px';el.style.overflowY='auto';}
      else{el.style.maxHeight='';el.style.overflowY='';}
    }
    // 锁定联动时同步滑块标签
    const oh=block.blockHeight;
    if(otherProp==='blockHeight'){
      const sr=document.getElementById('rs_h_'+id);
      if(sr)sr.value=oh;
      const sl=document.getElementById('sh_'+id);
      if(sl)sl.textContent=(oh&&oh!=='auto'&&oh!==0)?oh+'px':'自动';
    }else if(otherProp==='blockWidth'){
      const sr=document.getElementById('rs_w_'+id);
      if(sr)sr.value=block.blockWidth||100;
      const sl=document.getElementById('sw_'+id);
      if(sl)sl.textContent=(block.blockWidth||100)+'%';
    }
    // 锁定联动时图表也需要实时 resize
    if(block.type==='chart'){
      const wrap=el.querySelector('.chart-canvas-wrap');
      if(wrap){
        const ch=block.blockHeight||0;
        if(ch&&ch!==0&&ch!=='0'&&ch!=='auto'){wrap.style.height=ch+'px';}
        else{wrap.style.height='400px';}
      }
      if(chartInstances[id])chartInstances[id].resize();
    }
  }
}

function updateBlockSize(id,prop,value){
  const block=blocks.find(b=>b.id===id);
  if(!block)return;
  block[prop]=prop==='blockWidth'?parseInt(value):(parseInt(value)||'auto');
  // 锁定比例联动 — 同步另一维度数据和DOM
  const otherProp=syncLockedDimension(id,prop,value);
  // 同步另一滑块标签
  if(otherProp==='blockHeight'){
    const sr=document.getElementById('rs_h_'+id);
    const oh=block.blockHeight;
    if(sr)sr.value=(oh&&oh!=='auto'&&oh!==0)?oh:0;
    const sl=document.getElementById('sh_'+id);
    if(sl)sl.textContent=(oh&&oh!=='auto'&&oh!==0)?oh+'px':'自动';
  }else if(otherProp==='blockWidth'){
    const sr=document.getElementById('rs_w_'+id);
    if(sr)sr.value=block.blockWidth||100;
    const sl=document.getElementById('sw_'+id);
    if(sl)sl.textContent=(block.blockWidth||100)+'%';
  }
  const el=document.querySelector('.block[data-block-id="'+id+'"]');
  if(el){
    const w=block.blockWidth||100;
    el.style.width=w===100?w+'%':'calc('+w+'% - 6px)';
    const h=block.blockHeight;
    if(h&&h!==0&&h!=='0'&&h!=='auto'){el.style.maxHeight=h+'px';el.style.overflowY='auto';}
    else{el.style.maxHeight='';el.style.overflowY='';}
    // 图表高度同步 + resize
    if(block.type==='chart'){
      const wrap=el.querySelector('.chart-canvas-wrap');
      if(wrap&&h&&h!==0&&h!=='0'&&h!=='auto'){wrap.style.height=(h-80)+'px';}
      else if(wrap){wrap.style.height='';}
      if(chartInstances[id]){chartInstances[id].resize();}
    }
  }
  autoSave();
}

function toggleLockAspect(id,checked){
  const block=blocks.find(b=>b.id===id);
  if(!block)return;
  if(checked){
    const w=block.blockWidth||100;
    let h=block.blockHeight;
    // 自动设置默认高度
    if(!h||h==='auto'||h===0||h==='0'){
      h=Math.round((w/100)*400/20)*20; // 按比例默认400px基准
      if(h<40)h=40;
      block.blockHeight=h;
      // 更新高度滑块DOM
      const hs=document.getElementById('rs_h_'+id);
      if(hs)hs.value=h;
      const hl=document.getElementById('sh_'+id);
      if(hl)hl.textContent=h+'px';
    }
    block.lockAspect=true;
    block.aspectRatio=w/parseInt(h);
    toast('已锁定宽高比 '+w+'% : '+h+'px');
  }else{
    block.lockAspect=false;
    delete block.aspectRatio;
    toast('已取消锁定');
  }
  autoSave();
}

function syncLockedDimension(id,changedProp,newValue){
  const block=blocks.find(b=>b.id===id);
  if(!block||!block.lockAspect||!block.aspectRatio)return;
  const ratio=block.aspectRatio;
  let otherProp,otherValue;
  if(changedProp==='blockWidth'){
    otherProp='blockHeight';
    otherValue=Math.round(parseInt(newValue)/ratio/20)*20;
    if(otherValue<20)otherValue=20;
    if(otherValue>1200)otherValue=1200;
  }else{
    otherProp='blockWidth';
    otherValue=Math.round(parseInt(newValue)*ratio/5)*5;
    if(otherValue<25)otherValue=25;
    if(otherValue>100)otherValue=100;
  }
  block[otherProp]=otherValue;
  // Sync the UI slider and label
  const otherSlider=document.getElementById(otherProp==='blockWidth'?'rs_w_'+id:'rs_h_'+id);
  const otherLabel=document.getElementById(otherProp==='blockWidth'?'sw_'+id:'sh_'+id);
  if(otherSlider)otherSlider.value=otherValue;
  if(otherLabel){
    if(otherProp==='blockWidth')otherLabel.textContent=otherValue+'%';
    else otherLabel.textContent=otherValue===0?'自动':otherValue+'px';
  }
  return otherProp;
}

// 检测当前选区的样式激活状态（用于 toggle 判断）
// 关键：找到选区第一个文本节点的"直接父元素"，检查其 getComputedStyle
// getComputedStyle 已解析完整 CSS 层叠（含继承），直接父元素的值即为文本的生效样式
// 不能沿祖先链向上遍历——否则会被已覆盖的外层样式误导（如外层 bold + 内层 normal）
function isStyleActive(prop,activeValue){
  const sel=window.getSelection();
  if(!sel||sel.rangeCount===0)return false;
  const range=sel.getRangeAt(0);
  if(!range)return false;
  // 从 range.startContainer 开始，若不是文本节点就找其内的第一个文本节点
  let node=range.startContainer;
  if(node.nodeType===1){
    const walker=document.createTreeWalker(node,NodeFilter.SHOW_TEXT,null);
    const firstText=walker.nextNode();
    if(firstText)node=firstText;
  }
  if(!node)return false;
  if(node.nodeType===3)node=node.parentElement;
  if(!node)return false;
  // 只检查直接父元素的 computed style（已包含 CSS 继承和层叠）
  const cs=window.getComputedStyle(node);
  if(prop==='font-weight'){
    const fw=cs.fontWeight;
    if(fw==='bold'||fw==='bolder'||fw==='800'||fw==='900')return true;
    const num=parseInt(fw);
    if(!isNaN(num)&&num>=600)return true;
  }else if(prop==='text-decoration'){
    const v1=cs.textDecorationLine||'';
    const v2=cs.textDecoration||'';
    if(v1&&v1!=='none'&&v1.indexOf('underline')>=0)return true;
    if(v2&&v2!=='none'&&v2.indexOf('underline')>=0)return true;
  }else if(prop==='font-style'){
    const fs=cs.fontStyle;
    if(fs==='italic'||fs==='oblique')return true;
  }else{
    if(cs[prop]===activeValue)return true;
  }
  return false;
}

function formatCmd(cmd,val){
  // 立即保存当前选区（按钮 mousedown 先于父容器触发，不能等冒泡）
  saveSelectionFromToolbar();
  // 恢复选区
  if(savedRange){
    const sel=window.getSelection();
    sel.removeAllRanges();
    sel.addRange(savedRange);
  }
  // 加粗/斜体/下划线/字号/颜色 —— 全部用 CSS 内联 span 替代 execCommand
  // execCommand 在跨 span 元素（Ctrl+A 全选含已着色文字）时无法正确 toggle
  if(cmd==='bold'){
    const isBold=isStyleActive('font-weight','bold');
    applySpanStyle('font-weight',isBold?'normal':'bold');
  }else if(cmd==='italic'){
    const isItalic=isStyleActive('font-style','italic');
    applySpanStyle('font-style',isItalic?'normal':'italic');
  }else if(cmd==='underline'){
    const isUnderline=isStyleActive('text-decoration','underline');
    applySpanStyle('text-decoration',isUnderline?'none':'underline');
  }else if(cmd==='fontSize'){
    applySpanStyle('font-size',val);
  }else if(cmd==='foreColor'){
    applySpanStyle('color',val);
    addColorToHistory(val);  // 色谱选色也记录历史
  }else if(cmd==='insertLineBreak'){
    // 在选区末尾插入换行
    const sel=window.getSelection();
    if(sel.rangeCount>0){
      const r=sel.getRangeAt(0);
      r.collapse(false);
      const br=document.createElement('br');
      r.insertNode(br);
      // 移动光标到 br 后面
      r.setStartAfter(br);
      r.collapse(true);
      sel.removeAllRanges();
      sel.addRange(r);
    }
  }else{
    // 其他命令的兜底
    document.execCommand(cmd,false,val||null);
  }
  // 保存到数据 + 更新 savedRange
  if(formatTargetBlockId&&formatTargetEl&&document.contains(formatTargetEl)){
    const block=blocks.find(b=>b.id===formatTargetBlockId);
    if(block){
      if(formatTargetField){
        // 支持嵌套路径：sections.0 / events.0.desc / strengths / diag 等
        if(formatTargetField.indexOf('.')>-1){
          setNestedValue(block,formatTargetField,formatTargetEl.innerHTML);
        }else{
          block[formatTargetField]=formatTargetEl.innerHTML;
        }
      }else{
        block.html=formatTargetEl.innerHTML;
      }
      autoSave();
    }
  }
  // 更新保存的选区
  const sel=window.getSelection();
  if(sel&&sel.rangeCount>0){
    savedRange=sel.getRangeAt(0).cloneRange();
  }
}
function applySpanStyle(prop,value){
  const sel=window.getSelection();
  if(!sel||sel.rangeCount===0)return;
  const range=sel.getRangeAt(0);
  if(range.collapsed)return;
  // 核心算法：逐文本节点包裹
  // 关键：不能产生 <span><ul></ul></span> 这种 inline 包 block 的非法 HTML 结构
  // 浏览器会自动修复非法结构，强行闭合 span、插入零碎的空 block 元素/空行
  // 用户会看到"加粗后自动加了空行"——所以必须避免 inline span 跨越 block 元素边界
  //
  // 算法：手动递归收集 range 内所有文本节点 → 切割每个文本节点到 range 边界
  //       → 清除祖先链上已有的同属性 span（同属性的样式 + 解包空 span）
  //       → 包一个新的 inline span
  // 优点：保证每个 span 都只在 inline 文本节点上，不会跨 block 元素

  // 1. 收集 range 内的所有文本节点
  const textNodes=[];
  function visit(node){
    if(!node)return;
    if(node.nodeType===3){
      if(range.intersectsNode(node))textNodes.push(node);
      return;
    }
    for(let i=0;i<node.childNodes.length;i++)visit(node.childNodes[i]);
  }
  let startFrom=range.commonAncestorContainer;
  if(startFrom.nodeType!==1)startFrom=startFrom.parentNode;
  if(startFrom)visit(startFrom);
  if(textNodes.length===0)return;

  // 2. 对每个文本节点：切割 + 清除祖先链同属性 + 包新 span
  let firstTarget=null,lastTarget=null;
  textNodes.forEach(function(textNode){
    let startOffset=0;
    let endOffset=textNode.length;
    if(textNode===range.startContainer)startOffset=range.startOffset;
    if(textNode===range.endContainer)endOffset=range.endOffset;
    if(startOffset>=endOffset)return;

    let target=textNode;
    // 切割起点之前
    if(startOffset>0)target=target.splitText(startOffset);
    // 切割终点之后（只切 target 内的，不影响 textNode 整体）
    const newEnd=endOffset-startOffset;
    if(newEnd<target.length)target.splitText(newEnd);

    // 清除祖先链上同属性的 span（但不解包仍带其他样式的 span）
    let p=target.parentElement;
    while(p){
      if(p.tagName==='SPAN'&&p.style[prop]){
        p.style[prop]='';
        if(p.style.length===0){
          const pp=p.parentNode;
          if(pp){
            while(p.firstChild)pp.insertBefore(p.firstChild,p);
            pp.removeChild(p);
          }
        }
      }
      p=p.parentElement;
    }

    // 包新 span（仅当 value 不是"清除"语义时）
    if(value&&value!=='normal'&&value!=='none'){
      const span=document.createElement('span');
      span.style[prop]=value;
      target.parentNode.insertBefore(span,target);
      span.appendChild(target);
      if(!firstTarget)firstTarget=span;
      lastTarget=span;
    }else{
      // 清除值：仅记录位置用于恢复选区
      if(!firstTarget)firstTarget=target;
      lastTarget=target;
    }
  });

  // 3. 恢复选区
  if(firstTarget&&lastTarget){
    sel.removeAllRanges();
    const newRange=document.createRange();
    newRange.selectNodeContents(firstTarget);
    sel.addRange(newRange);
  }
}
// ===== 颜色快捷应用 + 历史记录 =====
const COLOR_HISTORY_KEY='bre_color_history';
const COLOR_HISTORY_MAX=8;
function getColorHistory(){
  try{
    const raw=localStorage.getItem(COLOR_HISTORY_KEY);
    return raw?JSON.parse(raw):[];
  }catch(e){return[];}
}
function addColorToHistory(color){
  if(!color)return;
  let hist=getColorHistory();
  // 去重：移除已存在的同名颜色
  hist=hist.filter(c=>c.toLowerCase()!==color.toLowerCase());
  // 加到最前
  hist.unshift(color);
  // 限制数量
  if(hist.length>COLOR_HISTORY_MAX)hist=hist.slice(0,COLOR_HISTORY_MAX);
  try{localStorage.setItem(COLOR_HISTORY_KEY,JSON.stringify(hist));}catch(e){}
  renderColorHistory();
}
function renderColorHistory(){
  const wrap=document.getElementById('ftRecent');
  if(!wrap)return;
  const hist=getColorHistory();
  if(hist.length===0){
    wrap.innerHTML='<span class="ft-clabel" style="color:#bbb">暂无</span>';
    return;
  }
  wrap.innerHTML=hist.map(c=>
    '<button class="ft-swatch" style="background:'+c+'" onmousedown="event.preventDefault();applyQuickColor(\''+c+'\')" title="'+c+'"></button>'
  ).join('');
}
function applyQuickColor(color){
  // 与 formatCmd('foreColor',color) 等效，但额外记录历史
  saveSelectionFromToolbar();
  if(savedRange){
    const sel=window.getSelection();
    sel.removeAllRanges();
    sel.addRange(savedRange);
  }
  applySpanStyle('color',color);
  addColorToHistory(color);
  // 保存到数据
  if(formatTargetBlockId&&formatTargetEl&&document.contains(formatTargetEl)){
    const block=blocks.find(b=>b.id===formatTargetBlockId);
    if(block){
      if(formatTargetField){
        if(formatTargetField.indexOf('.')>-1){
          setNestedValue(block,formatTargetField,formatTargetEl.innerHTML);
        }else{
          block[formatTargetField]=formatTargetEl.innerHTML;
        }
      }else{
        block.html=formatTargetEl.innerHTML;
      }
      autoSave();
    }
  }
  // 更新 savedRange
  const sel=window.getSelection();
  if(sel.rangeCount>0)savedRange=sel.getRangeAt(0).cloneRange();
}

function onTableRowNumClick(ri,ev){
  ev.preventDefault();ev.stopPropagation();
  const block=blocks.find(b=>b.id===editingChartBlockId);
  const nRows=((block&&block.tableData&&block.tableData.rows)||(block&&block.rows)||[]).length;
  // 切换：未选 → 选该行；已选 → 取消
  if(tableSel.selection==='row'&&Object.keys(tableSel.rows).length===1&&tableSel.rows[ri]){
    tableSel.selection='none';tableSel.rows={};
  }else{

function paintTableSelection(){
  // 清除所有 selected 样式
  document.querySelectorAll('#dataTable .cell-selected,#dataTable .row-selected,#dataTable .col-selected').forEach(el=>{
    el.classList.remove('cell-selected','row-selected','col-selected');
  });
  // 行高亮（变更背景）
  if(tableSel.selection==='row'){
    Object.keys(tableSel.rows).forEach(ri=>{
      const tr=document.querySelector('#dataTableBody tr:nth-child('+(parseInt(ri)+1)+')');
      if(tr){
        tr.querySelectorAll('.data-cell').forEach(td=>td.classList.add('cell-selected'));
        const rn=tr.querySelector('.row-num');if(rn)rn.classList.add('row-selected');
      }
    });
  }else if(tableSel.selection==='col'){
    Object.keys(tableSel.cols).forEach(ci=>{
      const idx=parseInt(ci)+1; // +1 for # column (querySelectorAll returns 0-based NodeList)
      document.querySelectorAll('#dataTable tr').forEach(tr=>{
        const tdList=tr.querySelectorAll('th,td');
        const cellTd=tdList[idx];
        if(cellTd){
          if(cellTd.tagName==='TH')cellTd.classList.add('col-selected');
          else if(cellTd.classList.contains('data-cell'))cellTd.classList.add('cell-selected');
        }
      });
    });
  }else if(tableSel.selection==='cell'){
    Object.keys(tableSel.cells).forEach(key=>{
      const [ri,ci]=key.split(',');
      const tr=document.querySelector('#dataTableBody tr:nth-child('+(parseInt(ri)+1)+')');
      if(tr){
        const tdList=tr.querySelectorAll('td');
        const cellTd=tdList[parseInt(ci)+1]; // +1 for # column
        if(cellTd)cellTd.classList.add('cell-selected');
      }
    });
  }
}
function updateTableSelectionStatus(){
  const el=document.getElementById('tableSelectionStatus');
  if(!el)return;
  if(tableSel.selection==='row'){
    el.textContent='已选整行（'+(Object.keys(tableSel.rows).map(r=>parseInt(r)+1).join('、'))+'）';
  }else if(tableSel.selection==='col'){
    el.textContent='已选整列（第 '+(Object.keys(tableSel.cols).map(c=>parseInt(c)+1).join('、'))+' 列）';
  }else if(tableSel.selection==='cell'){
    el.textContent='已选 '+Object.keys(tableSel.cells).length+' 个单元格';
  }else{
    el.textContent='未选择（点击下方表格单元格/行号/列头）';
  }
}

function markTableSelection(mode){
  const block=blocks.find(b=>b.id===editingChartBlockId);
  if(!block)return;
  if(!block.tableData)block.tableData={headers:block.headers||[],rows:block.rows||[]};
  const style=document.getElementById('tableMarkStyle').value;
  if(!block.cellMarks)block.cellMarks={};
  if(!block.rowMarks)block.rowMarks={};
  if(!block.colMarks)block.colMarks={};
  const rowsArr=block.tableData.rows||block.rows||[];
  const nCols=(block.tableData.headers||block.headers||[]).length;
  if(mode==='selected'){
    if(tableSel.selection==='row'){
      Object.keys(tableSel.rows).forEach(r=>block.rowMarks[r]=style);
    }else if(tableSel.selection==='col'){
      Object.keys(tableSel.cols).forEach(c=>block.colMarks[c]=style);
    }else if(tableSel.selection==='cell'){
      Object.keys(tableSel.cells).forEach(k=>block.cellMarks[k]=style);
    }else{
      toast('请先选择单元格、整行或整列');
      return;
    }
  }else if(mode==='row'){
    // 标记 selected 中所有行的整行（cells 里多行也算）
    const targetRows={};
    if(tableSel.selection==='row')Object.assign(targetRows,tableSel.rows);
    else if(tableSel.selection==='cell'){
      Object.keys(tableSel.cells).forEach(k=>{targetRows[k.split(',')[0]]=true;});
    }else if(tableSel.selection==='col'){
      // 整列模式 → 把列对应的所有行标记
      for(let r=0;r<rowsArr.length;r++)targetRows[r]=true;
    }else{
      for(let r=0;r<rowsArr.length;r++)targetRows[r]=true;
    }
    Object.keys(targetRows).forEach(r=>block.rowMarks[r]=style);
  }else if(mode==='col'){
    const targetCols={};
    if(tableSel.selection==='col')Object.assign(targetCols,tableSel.cols);
    else if(tableSel.selection==='cell'){
      Object.keys(tableSel.cells).forEach(k=>{targetCols[k.split(',')[1]]=true;});
    }else if(tableSel.selection==='row'){
      for(let c=0;c<nCols;c++)targetCols[c]=true;
    }else{
      for(let c=0;c<nCols;c++)targetCols[c]=true;
    }
    Object.keys(targetCols).forEach(c=>block.colMarks[c]=style);
  }else if(mode==='all'){
    for(let r=0;r<rowsArr.length;r++)block.rowMarks[r]=style;
    for(let c=0;c<nCols;c++)block.colMarks[c]=style;
  }
  // 重新渲染表格编辑器 DOM 来应用 class
  setTimeout(()=>openTableEditor(editingChartBlockId),30);
  // 同步刷新画布预览
  renderAll();
  autoSave();
}

// ===== Excel 导入 =====
function importExcelData(input){
  const file=input.files&&input.files[0];
  if(!file)return;
  const block=blocks.find(b=>b.id===editingChartBlockId);
  if(!block){input.value='';return;}
  const reader=new FileReader();
  reader.onload=function(e){
    try{
      const data=new Uint8Array(e.target.result);
      const wb=XLSX.read(data,{type:'array',cellDates:true});
      const ws=wb.Sheets[wb.SheetNames[0]];
      if(!ws){alert('Excel 文件中没有工作表');input.value='';return;}
      // 不设 defval，让合并单元格的非左上角位置保留为 undefined
      const aoaRaw=XLSX.utils.sheet_to_json(ws,{header:1,raw:false});
      if(aoaRaw.length===0){alert('Excel 工作表为空');input.value='';return;}
      // 跳过开头的完全空行
      let headerIdx=0;
      while(headerIdx<aoaRaw.length){
        const row=aoaRaw[headerIdx];
        if(row&&row.some(v=>v!==null&&v!==undefined&&String(v).trim()!==''))break;
        headerIdx++;
      }
      if(headerIdx>=aoaRaw.length){alert('Excel 中没有有效数据');input.value='';return;}
      // 智能识别表头：若第一行有合并单元格（含 undefined/null）且下一行非空值更多，则跳过第一行
      const firstRow=aoaRaw[headerIdx]||[];
      const secondRow=aoaRaw[headerIdx+1]||[];
      const firstHasMerges=firstRow.some(v=>v===null||v===undefined);
      const secondNonEmpty=secondRow.filter(v=>v!==null&&v!==undefined&&String(v).trim()!=='').length;
      if(firstHasMerges && secondNonEmpty>=2){
        headerIdx++;
      }
      // 计算最大列数（取所有非空行中最右非空位置+1）
      let maxCols=0;
      for(let i=headerIdx;i<aoaRaw.length;i++){
        const row=aoaRaw[i]||[];
        let lastIdx=-1;
        for(let j=row.length-1;j>=0;j--){
          if(row[j]!==null&&row[j]!==undefined&&String(row[j]).trim()!==''){
            lastIdx=j;break;
          }
        }
        if(lastIdx>=0)maxCols=Math.max(maxCols,lastIdx+1);
      }
      if(maxCols===0){alert('Excel 中未检测到数据');input.value='';return;}
      // 提取表头（按 maxCols 截取/补空）
      const headerRow=aoaRaw[headerIdx]||[];
      const headers=[];
      for(let c=0;c<maxCols;c++){
        const v=headerRow[c];
        if(v===null||v===undefined)headers.push('');
        else headers.push(String(v).trim());
      }
      // 提取数据行（表头之后的所有非空行，按 maxCols 对齐）
      const dataRows=[];
      for(let i=headerIdx+1;i<aoaRaw.length;i++){
        const row=aoaRaw[i]||[];
        // 跳过完全空行
        let hasAny=false;
        for(let j=0;j<row.length;j++){
          if(row[j]!==null&&row[j]!==undefined&&String(row[j]).trim()!==''){hasAny=true;break;}
        }
        if(!hasAny)continue;
        const r=[];
        for(let c=0;c<maxCols;c++){
          const v=row[c];
          r.push(v!==null&&v!==undefined?String(v):'');
        }
        dataRows.push(r);
      }
      block.tableData={headers:headers,rows:dataRows};
      block.headers=headers;
      block.rows=dataRows;
      // 同步 headerLevels
      if(block.headerRows&&block.headerRows>1){
        const levels=[];
        for(let i=0;i<block.headerRows;i++){
          levels.push(headers.map(h=>({text:i===0?h:'',rowspan:1,colspan:1})));
        }
        block.headerLevels=levels;
      }
      // 刷新表格编辑器
      openTableEditor(editingChartBlockId);
      // 显示成功提示
      var tip=document.createElement('div');
      tip.textContent='✅ 已导入 '+dataRows.length+' 行 × '+maxCols+' 列';
      tip.style.cssText='position:fixed;top:20px;left:50%;transform:translateX(-50%);background:#10b981;color:#fff;padding:10px 24px;border-radius:8px;font-size:14px;font-weight:600;z-index:10000;box-shadow:0 4px 12px rgba(0,0,0,.2)';
      document.body.appendChild(tip);
      setTimeout(()=>{tip.style.transition='opacity .4s';tip.style.opacity='0';setTimeout(()=>tip.remove(),400);},2500);
    }catch(err){
      alert('导入失败：'+err.message);
    }
    input.value='';
  };
  reader.onerror=function(){alert('文件读取失败');input.value='';};
  reader.readAsArrayBuffer(file);
}

// ===== 表格条件格式 / 多层表头 / 合并单元格辅助 =====
function updateTableHeaderRows(id,n){
  const block=blocks.find(b=>b.id===id);
  if(!block)return;
  n=Math.max(1,Math.min(3,n||1));
  block.headerRows=n;
  // 同步初始化 headerLevels
  const labels=block.headers||[];
  if(!block.headerLevels||block.headerLevels.length!==n){
    const newLevels=[];
    for(let i=0;i<n;i++){
      newLevels.push(labels.map(h=>({text:i===0?(h||''):'',rowspan:1,colspan:1})));
    }
    block.headerLevels=newLevels;
  }
  renderAll();
  updateRightPanel();
  autoSave();
}

function updateTableCF(id,colIdx,rule){
  const block=blocks.find(b=>b.id===id);
  if(!block)return;
  if(!block.conditionalFormat)block.conditionalFormat={};
  if(!rule||rule===''){
    delete block.conditionalFormat[colIdx];
  }else{
    block.conditionalFormat[colIdx]=rule;
  }
  renderAll();
  autoSave();
}

function updateTableMergeCell(id,rowIdx,colIdx,prop,val){
  const block=blocks.find(b=>b.id===id);
  if(!block)return;
  if(!block.headerLevels)updateTableHeaderRows(id,block.headerRows||1);
  if(!block.headerLevels[rowIdx])block.headerLevels[rowIdx]=[];
  if(!block.headerLevels[rowIdx][colIdx])block.headerLevels[rowIdx][colIdx]={text:'',rowspan:1,colspan:1};
  block.headerLevels[rowIdx][colIdx][prop]=val;
  // 仅记录，不立即 render（用户可能连续编辑多格）
  if(prop==='text'){
    // 文本变化只更新画布，不重绘右侧面板
    renderAll();
  }
  autoSave();
}

function applyTableMerges(id){
  const block=blocks.find(b=>b.id===id);
  if(!block)return;
  // headerLevels 已是数据源
  // 也把第一层表头同步到 headers（兼容老逻辑）

    block.headers=block.headerLevels[0].map(c=>typeof c==='object'?(c.text||''):c);
  }
  renderAll();
  updateRightPanel();
  autoSave();
  toast('表头合并已应用');
}

let draggedBlockId=null;
function handleDragStart(e,id){
  draggedBlockId=id;
  e.target.classList.add('dragging');
  e.dataTransfer.effectAllowed='move';
  e.dataTransfer.setData('text/plain',id);
}
function handleDragEnd(e){
  e.target.classList.remove('dragging');
  document.querySelectorAll('.block.drag-over').forEach(el=>el.classList.remove('drag-over'));
  draggedBlockId=null;
}
function handleDragOver(e){
  e.preventDefault();
  e.dataTransfer.dropEffect='move';
}
function handleDrop(e){
  e.preventDefault();
  const dropTarget=e.target.closest('.block');
  if(!dropTarget||!draggedBlockId||draggedBlockId===dropTarget.dataset.blockId)return;
  const fromIdx=blocks.findIndex(b=>b.id===draggedBlockId);
  const toIdx=blocks.findIndex(b=>b.id===dropTarget.dataset.blockId);
  if(fromIdx<0||toIdx<0)return;
  const [item]=blocks.splice(fromIdx,1);
  blocks.splice(toIdx,0,item);
  renderAll();
  autoSave();
  toast('已调整顺序');
}

function renderTableBlockForExport(block){
  const title=escHtml(block.title||'');
  const headers=block.headers||[];
  const rows=block.rows||[];
  const headerRows=block.headerRows||1;
  const headerLevels=block.headerLevels||null;
  const merges=block.merges||{};
  // 手动标记：cellMarks / rowMarks / colMarks
  const cellMarks=block.cellMarks||{};
  const rowMarks=block.rowMarks||{};
  const colMarks=block.colMarks||{};
  let headHtml='';
  if(headerLevels&&headerLevels.length>0){
    for(let hr=0;hr<headerLevels.length;hr++){
      headHtml+='<tr>';
      headerLevels[hr].forEach(function(cell,ci){
        const m=merges[hr+','+ci];
        let attrs='';
        let txt=cell;
        if(typeof cell==='object'&&cell!==null){
          txt=cell.text||'';
          if(cell.rowspan>1)attrs+=' rowspan="'+cell.rowspan+'"';
          if(cell.colspan>1)attrs+=' colspan="'+cell.colspan+'"';
        }else{if(m){if(m.rowspan>1)attrs+=' rowspan="'+m.rowspan+'"';if(m.colspan>1)attrs+=' colspan="'+m.colspan+'"';}}
        headHtml+='<th class="merged-cell"'+attrs+'>'+escHtml(txt)+'</th>';
      });
      headHtml+='</tr>';
    }
  }else{
    headHtml+='<tr>'+headers.map(function(h,i){
      const m=merges['0,'+i];
      let attrs='';
      if(m){if(m.rowspan>1)attrs+=' rowspan="'+m.rowspan+'"';if(m.colspan>1)attrs+=' colspan="'+m.colspan+'"';}
      return '<th class="merged-cell"'+attrs+'>'+escHtml(h)+'</th>';
    }).join('')+'</tr>';
  }
  let bodyHtml='';
  rows.forEach(function(row,ri){
    bodyHtml+='<tr>';
    for(let ci=0;ci<headers.length;ci++){
      const mkey=(ri+headerRows)+','+ci;
      const m=merges[mkey];
      let attrs='';
      if(m){if(m.rowspan>1)attrs+=' rowspan="'+m.rowspan+'"';if(m.colspan>1)attrs+=' colspan="'+m.colspan+'"';if(m.skip)continue;}
      // 手动标记：cell > row > col
      const mark=cellMarks[ri+','+ci]||rowMarks[ri]||colMarks[ci]||'';
      let cellContent=escHtml(row[ci]!==undefined?row[ci]:'');
      if(mark){cellContent='<span class="cell-mark-'+mark+'">'+cellContent+'</span>';}
      let tdClass='';
      if(cellMarks[ri+','+ci])tdClass=' cell-mark cell-mark-'+cellMarks[ri+','+ci];
      else if(rowMarks[ri])tdClass=' cell-mark cell-mark-row-'+rowMarks[ri];
      else if(colMarks[ci])tdClass=' cell-mark cell-mark-col-'+colMarks[ci];
      bodyHtml+='<td'+attrs+tdClass+'>'+cellContent+'</td>';
    }
    bodyHtml+='</tr>';
  });
  return '<h3 style="margin-bottom:8px">'+title+'</h3><div class="table-wrap"><table><thead>'+headHtml+'</thead><tbody>'+bodyHtml+'</tbody></table></div>';
}

// ===== 保存/加载工程 =====
function saveProject(){
  const data=JSON.stringify({sections:sections,blocks:blocks,currentSectionId:currentSectionId},null,2);
  const name=prompt('工程名称：','品牌复盘项目');
  if(!name)return;
  localStorage.setItem('report_project_'+name,data);
  localStorage.setItem('report_project_last',name);
  toast('工程已保存: '+name);
}

function loadProject(){
  const last=localStorage.getItem('report_project_last')||'';
  const name=prompt('工程名称（输入已保存的名称）：',last);
  if(!name)return;
  const raw=localStorage.getItem('report_project_'+name);
  if(!raw){alert('未找到工程: '+name);return;}
  try{
    const data=JSON.parse(raw);
    // 兼容老格式（直接是 blocks 数组）
    if(Array.isArray(data)){
      blocks=data;
      sections=[{id:'sec_default',name:'首页'}];
      currentSectionId='sec_default';
    }else{
      blocks=data.blocks||[];
      sections=data.sections&&data.sections.length>0?data.sections:[{id:'sec_default',name:'首页'}];
      currentSectionId=data.currentSectionId||(sections[0]&&sections[0].id)||'sec_default';
    }
    Object.values(chartInstances).forEach(c=>c.destroy());
    Object.keys(chartInstances).forEach(k=>delete chartInstances[k]);
    selectedBlockId=null;
    renderAll();
    localStorage.setItem('report_project_last',name);
    toast('工程已加载: '+name);
  }catch(e){alert('工程数据损坏');}
}

function resetAll(){
  if(!confirm('确认重置？所有未保存的内容将丢失。'))return;
  Object.values(chartInstances).forEach(c=>c.destroy());
  Object.keys(chartInstances).forEach(k=>delete chartInstances[k]);
  blocks=[];
  sections=[{id:'sec_default',name:'首页'}];
  currentSectionId='sec_default';
  selectedBlockId=null;
  renderAll();
  toast('已重置');
}

function downloadFile(filename,content){
  const blob=new Blob([content],{type:'text/html;charset=utf-8'});
  const url=URL.createObjectURL(blob);
  const a=document.createElement('a');
  a.href=url;a.download=filename;
  document.body.appendChild(a);a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
}
